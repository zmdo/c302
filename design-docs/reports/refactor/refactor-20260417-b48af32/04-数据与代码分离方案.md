**新代码模式**：

```python
# c302/parameters/loader.py
class ParameterLoader:
    """YAML 参数集加载器。"""

    def load(self, level: str) -> "ParameterisedModel":
        """加载指定层级的参数集。"""
        yaml_path = get_data_dir() / "parameters" / f"level_{level.lower()}.yaml"
        with open(yaml_path) as f:
            data = yaml.safe_load(f)

        model = ParameterisedModel(level=data["level"])
        for param in data["parameters"]:
            model.add_bioparameter(
                param["name"], param["value"],
                param["source"], param["certainty"]
            )
        return model
```

### 4. 参数继承方案

对于 C0/C1 这类基于 C 的变体，YAML 支持继承：

```yaml
# data/parameters/level_c0.yaml
level: "C0"
description: "C 的简化变体（无 fast K 通道）"
inherits: "level_c"  # 继承 C 的全部参数

# 仅列出差异参数（覆盖继承值）
parameter_overrides:
  - name: "k_fast_max_conductance"
    value: "0 nS"
    source: "C0 disables fast K"
    certainty: "1.0"

# 仅列出需要删除的参数
parameter_removals:
  - "k_fast_*"  # 支持通配符
```

加载器处理继承逻辑：

```python
def load(self, level: str) -> "ParameterisedModel":
    data = self._load_yaml(level)

    # 处理继承
    if "inherits" in data:
        base_model = self.load(data["inherits"].replace("level_", ""))
        # 应用覆盖
        for override in data.get("parameter_overrides", []):
            base_model.set_bioparameter(override["name"], override["value"], ...)
        # 应用删除
        for removal in data.get("parameter_removals", []):
            base_model.remove_bioparameters(removal)
        return base_model

    # 常规加载
    ...
```

## 数据完整性校验

新代码应在启动时校验关键数据文件的存在性：

```python
# c302/utils/helpers.py
def validate_data_integrity() -> None:
    """校验数据文件完整性。"""
    required_files = [
        get_data_dir() / "connectome" / "neurons.yaml",
        get_data_dir() / "connectome" / "muscles.yaml",
    ]
    # 校验参数文件
    for level in ["a", "b", "bc1", "c", "c0", "c1", "c2", "d", "d1", "w2d"]:
        required_files.append(get_data_dir() / "parameters" / f"level_{level}.yaml")

    missing = [f for f in required_files if not f.exists()]
    if missing:
        raise FileNotFoundError(f"缺少数据文件: {missing}")
```
