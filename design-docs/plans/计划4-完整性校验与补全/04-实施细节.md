```python
"""用原始 c302 代码生成 72 组参考基线数据。"""
import json
import sys
import os

# 将原始代码路径加入 sys.path
sys.path.insert(0, os.path.join(os.path.dirname(__file__), ".."))

LEVELS = ["A", "B", "BC1", "C0", "C", "C1", "C2", "D", "D1", "W2D"]
CONFIGS = ["IClamp", "Syns", "Social", "Pharyngeal", "Full", "Muscles", "Oscillator", "FW"]

def extract_baseline(nml_doc, params, config, level):
    """从生成的 NeuroML 文档提取基线计数数据。"""
    net = nml_doc.networks[0]
    # 种群
    pop_names = [p.id for p in net.populations]
    # 连接：统计 projections + electrical_projections + continuous_projections
    chem = sum(len(p.connections) + len(p.connection_wds) for p in net.projections)
    elec = sum(
        len(p.electrical_connections) + len(p.electrical_connection_instances)
        + len(getattr(p, "electrical_connection_instance_ws", []))
        for p in net.electrical_projections
    )
    cont = sum(
        len(p.continuous_connections) + len(p.continuous_connection_instances)
        + len(getattr(p, "continuous_connection_instance_ws", []))
        for p in net.continuous_projections
    )
    # 刺激
    stimuli = len(net.input_lists)
    stim_count = sum(len(il.input) for il in net.input_lists)
    # BioParameter
    bp_count = len(params.bioparameters)
    bp_values = {bp.name: bp.value for bp in params.bioparameters}
    # 细胞模型
    iaf = len(nml_doc.iaf_cells)
    hh = len(nml_doc.cells)
    # 突触模型
    exp_two = len(nml_doc.exp_two_synapses)
    gap = len(nml_doc.gap_junctions)
    graded = len(nml_doc.graded_synapses)
    graded2 = len(getattr(nml_doc, "graded_synapses2", []))

    return {
        "config": config,
        "level": level,
        "populations": {"count": len(pop_names), "names": sorted(pop_names)},
        "connections": {"chemical": chem, "electrical": elec, "continuous": cont, "total": chem + elec + cont},
        "stimuli": {"input_lists": stimuli, "total_inputs": stim_count},
        "bioparameters": {"count": bp_count, "values": bp_values},
        "cell_models": {"iaf_cells": iaf, "hh_cells": hh},
        "synapse_models": {"exp_two_synapses": exp_two, "gap_junctions": gap, "graded_synapses": graded, "graded_synapses2": graded2},
    }
```

            conductance=self._bp("neuron_to_neuron_exc_syn_conductance"),

            erev=self._bp("exc_syn_erev"),
            k=self._bp("exc_syn_k"),
        )
        self.neuron_to_neuron_inh_syn = GradedSynapse(

            delta=self._bp("inh_syn_delta"),
            Vth=self._bp("inh_syn_vth"),
            erev=self._bp("inh_syn_erev"),
            k=self._bp("inh_syn_k"),
        )
        # 电突触
        self.neuron_to_neuron_elec_syn = GapJunction(
            id="neuron_to_neuron_elec_syn",

        )

    def create_neuron_to_muscle_syn(self) -> None:
        """创建神经元到肌肉突触——GradedSynapse + GapJunction。"""
        # 与 n2n 共享 delta/vth/erev/k 参数
        self.neuron_to_muscle_exc_syn = GradedSynapse(
            id="neuron_to_muscle_exc_syn",
            conductance=self._bp("neuron_to_muscle_exc_syn_conductance"),
            delta=self._bp("exc_syn_delta"),
            Vth=self._bp("exc_syn_vth"),
            erev=self._bp("exc_syn_erev"),
            k=self._bp("exc_syn_k"),
        )
        self.neuron_to_muscle_inh_syn = GradedSynapse(...)
        self.neuron_to_muscle_elec_syn = GapJunction(...)

    def get_exc_syn(self, pre, post, type_) -> GradedSynapse:
        """兴奋性突触——返回对应 GradedSynapse 原型。"""
        if type_ == "neuron_to_muscle":
            return self.neuron_to_muscle_exc_syn
        return self.neuron_to_neuron_exc_syn

    def get_inh_syn(self, pre, post, type_) -> GradedSynapse:
        """抑制性突触——返回对应 GradedSynapse 原型。"""
        if type_ == "neuron_to_muscle":
            return self.neuron_to_muscle_inh_syn
        return self.neuron_to_neuron_inh_syn

    def get_elec_syn(self, pre, post, type_) -> GapJunction:
        """电突触——返回 GapJunction。"""
        params = self._get_elec_syn_params(type_)
        return GapJunction(id=f"{type_}_elec_syn", conductance=params["gbase"])

        self.generic_neuron_cell = CellW2D(id="GenericNeuronCell")

        # W2D: 神经元到肌肉仅有兴奋性 + 电
        self.neuron_to_muscle_exc_syn = OutputSynapse(id="neuron_to_muscle_w2d")
        self.neuron_to_muscle_elec_syn = GapJunction(
            id="neuron_to_muscle_elec_syn",
            conductance=self._bp("neuron_to_muscle_elec_syn_gbase"),
        )
        # 无抑制性突触
        self.neuron_to_muscle_inh_syn = None

    def create_muscle_to_muscle_syn(self) -> None:
        """W2D 特有：肌肉间突触。"""
        self.muscle_to_muscle_exc_syn = OutputSynapse(id="muscle_to_muscle_w2d")

            id="muscle_to_muscle_elec_syn",
            conductance=self._bp("muscle_to_muscle_elec_syn_gbase"),

        gbase = self._bp(f"{type_}_elec_syn_gbase")
        return GapJunction(id=f"{type_}_elec_syn", conductance=gbase)

    def get_exc_syn(self, pre, post, type_):
        return self.neuron_to_neuron_exc_syn

    def get_inh_syn(self, pre, post, type_):
        return self.neuron_to_neuron_inh_syn
