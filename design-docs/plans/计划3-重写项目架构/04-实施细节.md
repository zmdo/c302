### `project_r1/c302/utils/helpers.py` — 路径管理

```python
# =============================================================================
# 功能描述：
#   通用工具函数，包含项目路径管理和格式化工具。
# ...
# =============================================================================
import logging
from pathlib import Path

logger = logging.getLogger(__name__)


def get_project_root() -> Path:
    """获取项目根目录（包含 pyproject.toml 的目录）。

    :return: 项目根目录路径
    :raises FileNotFoundError: 无法找到 pyproject.toml
    """
    # 从当前文件向上查找包含 pyproject.toml 的目录
    current = Path(__file__).resolve().parent
    while current != current.parent:
        if (current / "pyproject.toml").exists():
            return current
        current = current.parent
    raise FileNotFoundError("无法找到项目根目录")


def get_data_dir() -> Path:
    """获取数据目录路径。

    :return: data/ 目录路径
    """
    return get_project_root() / "data"


def get_morphology_dir() -> Path:
    """获取细胞形态文件目录。

    :return: data/morphology/ 目录路径
    """
    return get_data_dir() / "morphology"


def get_xml_dir() -> Path:
    """获取 XML 资源目录。

    :return: data/xml/ 目录路径
    """
    return get_data_dir() / "xml"
```

### `project_r1/c302/parameters/bio.py` — 参数类

```python
# =============================================================================
# 功能描述：
#   BioParameter 数据类定义，表示单个生物物理参数。
# ...
# =============================================================================
from decimal import Decimal
from dataclasses import dataclass


@dataclass
class BioParameter:
    """生物物理参数。

    :param name: 参数名称
    :param value: 参数值（含单位的字符串）
    :param source: 参数来源说明
    :param certainty: 确定性等级
    """
    name: str
    value: str
    source: str
    certainty: str

    def change_magnitude(self, factor: float) -> "BioParameter":
        """按倍率修改参数值，保留单位不变。

        :param factor: 倍率因子
        :return: 修改后的 BioParameter（原地修改并返回）
        """
        # 分离数值和单位部分
        val_str = self.value.strip()
        i = 0
        for i, ch in enumerate(val_str):
            if ch.isalpha():
                break
        else:
            i = len(val_str)

        # 用 Decimal 保持精度
        num = Decimal(val_str[:i].strip())
        unit = val_str[i:].strip()
        new_num = num * Decimal(str(factor))
        self.value = f"{new_num} {unit}".strip()
        return self

    @property
    def x(self) -> float:
        """以 float 形式返回参数的数值部分（去除单位）。

        :return: 数值
        """
        val_str = self.value.strip()
        num_str = ""
        for ch in val_str:
            if ch in "0123456789.eE+-":
                num_str += ch
            elif num_str:
                break
        return float(num_str) if num_str else 0.0
```

### `project_r1/c302/parameters/loader.py` — YAML 加载器

```python
# =============================================================================
# 功能描述：
#   YAML 参数集加载器，支持参数继承和覆盖机制。
# ...
# =============================================================================
import logging
from pathlib import Path

import yaml

from c302.parameters.bio import BioParameter
from c302.utils.helpers import get_data_dir

logger = logging.getLogger(__name__)


class ParameterLoader:
    """YAML 参数集加载器。"""

    def __init__(self) -> None:
        """初始化加载器，创建缓存字典。"""
        self._cache: dict[str, dict] = {}

    def load_raw(self, level: str) -> dict:
        """加载原始 YAML 数据，带缓存。

        :param level: 参数层级名称（如 "A", "C0"）
        :return: YAML 解析后的字典
        """
        # 规范化文件名
        filename = f"level_{level.lower()}.yaml"
        if filename in self._cache:
            return self._cache[filename]

        yaml_path = get_data_dir() / "parameters" / filename
        with open(yaml_path, encoding="utf-8") as f:
            data = yaml.safe_load(f)

        self._cache[filename] = data
        return data

    def load_parameters(self, level: str) -> list[BioParameter]:
        """加载指定层级的参数集，处理继承逻辑。

        :param level: 参数层级名称（如 "A", "C0"）
        :return: BioParameter 列表
        """
        data = self.load_raw(level)

        # 处理继承：先加载基类参数
        if "inherits" in data:
            base_level = data["inherits"].replace("level_", "")
            params = self.load_parameters(base_level)
            param_dict = {p.name: p for p in params}

            # 应用覆盖
            for override in data.get("parameter_overrides", []):
                bp = BioParameter(
                    name=override["name"],
                    value=override["value"],
                    source=override["source"],
                    certainty=override["certainty"],
                )
                param_dict[bp.name] = bp

            # 应用删除
            for removal in data.get("parameter_removals", []):
                # 支持通配符匹配
                if "*" in removal:
                    prefix = removal.replace("*", "")
                    to_remove = [k for k in param_dict if k.startswith(prefix)]
                    for k in to_remove:
                        del param_dict[k]
                elif removal in param_dict:
                    del param_dict[removal]

            return list(param_dict.values())

        # 常规加载
        return [
            BioParameter(
                name=p["name"],
                value=p["value"],
                source=p["source"],
                certainty=p["certainty"],
            )
            for p in data.get("parameters", [])
        ]
```

## 编码规范要求

所有新写代码必须遵循 SP-CODE-2026-001，具体包括：

1. **文件头**：每个 .py 文件包含功能描述、类与方法索引、更新日志、当前维护者
2. **reST docstring**：所有公开模块、类、函数和方法
3. **类型注解**：所有函数参数和返回值
4. **强制注释**：函数体内关键逻辑、条件分支、循环、异常处理均需中文行内注释
5. **命名规范**：模块 snake_case、类 PascalCase、常量 UPPER_SNAKE_CASE
6. **双引号**：统一使用 `"` 包裹字符串
7. **日志**：使用 `logging` 模块，禁止 `print()`
8. **单元测试**：每个模块对应 `test_*.py`，每个 public 函数至少正向 + 异常各 1 个测试
