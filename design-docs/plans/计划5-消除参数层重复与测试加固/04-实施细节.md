## 1. factory/__init__.py

```python
# =============================================================================
# 功能描述：
#   参数工厂入口模块，提供 create_model() 工厂函数。
#   根据层级名称创建对应的参数化模型实例。
#
# 类与方法索引：
#   create_model                         — 工厂函数，按层级名创建模型实例
#
# 更新日志：
#   2026-04-19  Copilot  计划5：从 factory.py 拆分
#
# 当前维护者：Copilot
# =============================================================================
"""参数工厂入口。"""
import logging

from c302.parameters.model import c302ModelPrototype

logger = logging.getLogger(__name__)


def create_model(level: str) -> c302ModelPrototype:
    """根据层级名称创建参数化模型实例。

    :param level: 层级名称（如 "A"、"C0"、"D1"）
    :return: c302ModelPrototype 实例
    :raises ValueError: 未知的层级名称
    """
    # 延迟导入具体类，避免循环依赖
    from c302.parameters.factory.iaf import (
        _BC1Model,
        _IafActivityModel,
        _IafModel,
    )
    from c302.parameters.factory.hh import (
        _HHC0Model,
        _HHC1Model,
        _HHModel,
    )
    from c302.parameters.factory.hh_multi import (
        _HHGradedModel,
        _HHMultiCompModel,
    )
    from c302.parameters.factory.special import _C2Model, _W2DModel

    _LEVEL_TO_CLASS: dict[str, type] = {
        "A": _IafModel,
        "B": _IafActivityModel,
        "BC1": _BC1Model,
        "C": _HHModel,
        "C0": _HHC0Model,
        "C1": _HHC1Model,
        "C2": _C2Model,
        "D": _HHMultiCompModel,
        "D1": _HHGradedModel,
        "W2D": _W2DModel,
    }

    level_upper = level.upper()
    cls = _LEVEL_TO_CLASS.get(level_upper)
    if cls is None:
        raise ValueError(
            f"未知的参数层级: {level!r}，"
            f"可用: {sorted(_LEVEL_TO_CLASS.keys())}"
        )
    logger.info("创建 %s 层级模型: %s", level_upper, cls.__name__)
    return cls(level_upper)
```

### 辅助方法 `_build_syn_from_params`

```python
def _build_syn_from_params(
    self,
    syn_cls: type,
    template: str,
    param_fields: tuple[str, ...],
    param_to_kwarg: dict[str, str] | None = None,
) -> object:
    """根据参数字段从 BioParameter 查找值并构造突触对象。

    :param syn_cls: 突触类（如 ExpTwoSynapse, GradedSynapse）
    :param template: 参数名模板（如 "neuron_to_neuron_chem_exc_syn_%s"）
    :param param_fields: 参数字段名列表（如 ("gbase", "erev", "decay", "rise")）
    :param param_to_kwarg: 字段名 → 构造参数名映射（如 {"decay": "tau_decay"}）
    :return: 突触对象实例
    """
    kwargs = {}
    for field in param_fields:
        bp = self.get_bioparameter(template % field)
        if bp:
            # 映射字段名到构造参数名
            kwarg_name = field
            if param_to_kwarg and field in param_to_kwarg:
                kwarg_name = param_to_kwarg[field]
            kwargs[kwarg_name] = bp.value
    # id 为模板去掉末尾 _%s
    syn_id = template.replace("_%s", "").replace("chem_", "")
    return syn_cls(id=syn_id, **kwargs)
```

### 模板方法 `get_exc_syn` / `get_inh_syn`

保持当前 `_ModelBase` 中的 `ExpTwoSynapse` 实现为默认。子类通过设置类属性切换行为：

- `_BC1Model`、`_HHC1Model`：设置 `_exc_syn_cls = GradedSynapse`，字段 `("conductance", "erev", "delta", "vth", "k")`
- `_HHC0Model`、`_HHGradedModel`：设置 `_exc_syn_cls = GradedSynapse2`，字段 `("conductance", "erev", "ar", "ad", "beta", "vth")`

`get_exc_syn` / `get_inh_syn` 在 `_ModelBase` 中使用 `self._exc_syn_cls` 和 `self._exc_param_fields` 分派，而非硬编码 `ExpTwoSynapse`。`_C2Model` 和 `_W2DModel` 保留 override。

### `_GradedSynapse2Mixin`

```python
class _GradedSynapse2Mixin:
    """为使用 GradedSynapse2 的模型提供共享方法。"""

    def create_n_connection_synapse(
        self, prototype_syn, n, nml_doc, existing_synapses
    ):
        """创建 N 连接突触实例。"""
        # 调用父类方法（处理 ExpTwoSynapse / GapJunction）
        result = super().create_n_connection_synapse(
            prototype_syn, n, nml_doc, existing_synapses
        )
        if result is not None:
            return result
        # 处理 GradedSynapse2
        if isinstance(prototype_syn, GradedSynapse2):
            new_syn = GradedSynapse2(...)  # 复制并修改 id
            nml_doc.graded_synapses.append(new_syn)
            return new_syn
        return None

    def is_analog_conn(self, syn) -> bool:
        """判断是否为模拟连接。"""
        return super().is_analog_conn(syn) or isinstance(syn, GradedSynapse2)
```

## 7. configs/__init__.py 修改

替换 `_auto_import()` 实现：

```python
def _auto_import() -> None:
    """自动导入 configs/ 目录下所有配置模块以触发注册。"""
    import importlib
    import pkgutil

    # 获取 configs 包路径
    package = importlib.import_module("c302.configs")
    # 扫描目录下所有非下划线开头的模块
    for _importer, modname, ispkg in pkgutil.iter_modules(package.__path__):
        if not ispkg and not modname.startswith("_"):
            try:
                importlib.import_module(f"c302.configs.{modname}")
            except ImportError:
                logger.warning("无法导入配置模块: c302.configs.%s", modname)
```

## 8. test_equivalence.py xfail 细化

### 变更要点

1. 删除 `_mark_known_diffs()` 函数
2. 新增两个参数列表：
   - `BASELINE_PARAMS`：无 xfail，用于大多数测试方法
   - `BASELINE_PARAMS_VALUES`：仅对 Muscles/Oscillator 组合标记 xfail
   - `BASELINE_PARAMS_MUSCLES_STRUCT`：仅对 Muscles 组合标记 xfail（`test_count_exact`/`test_total_exact`）

```python
# 无 xfail 的参数列表（用于大多数测试）
BASELINE_PARAMS = [pytest.param(*c) for c in BASELINE_CASES]

# values 测试的 xfail 参数列表（Muscles + Oscillator 的 test_values_match）
BASELINE_PARAMS_VALUES = [
    pytest.param(
        *c,
        marks=pytest.mark.xfail(
            reason=f"{c[0]} config 级参数覆盖差异",
            strict=False,
        ),
    )
    if c[0] in _KNOWN_CONFIG_DIFFS_CONFIGS
    else pytest.param(*c)
    for c in BASELINE_CASES
]

# Muscles 结构测试的 xfail 参数列表（仅 Muscles 的 count_exact/total_exact）
BASELINE_PARAMS_MUSCLES_STRUCT = [
    pytest.param(
        *c,
        marks=pytest.mark.xfail(
            reason=f"{c[0]} config 级参数覆盖导致种群/连接数差异",
            strict=False,
        ),
    )
    if c[0] == "Muscles"
    else pytest.param(*c)
    for c in BASELINE_CASES
```

5. `TestPopulations.test_count_exact` 和 `TestConnections.test_total_exact` 使用 `BASELINE_PARAMS_MUSCLES_STRUCT`
